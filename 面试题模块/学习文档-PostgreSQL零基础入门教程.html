PostgreSQL 从入门到精通

本文档从零开始系统讲解 PostgreSQL 关系数据库，适合后端开发者、数据工程师、以及想系统掌握 PG 技术的开发者。从基础概念讲到生产部署，SQL 全部可运行。

一、PostgreSQL 基础概念

1.1 为什么选 PostgreSQL

PostgreSQL（简称 PG）是全球最先进的开源关系数据库，30 年持续开发。

维度 PostgreSQL MySQL
JSON 支持 JSONB（二进制）+ GIN 索引 JSON（文本存储，查询慢）
索引类型 B-tree, GIN, GiST, SP-GiST, BRIN, Hash B-tree, Hash
窗口函数 支持完善 8.0+ 支持
CTE 递归 ✅ 支持 8.0+ 支持
全文搜索 内置 需外部方案
扩展生态 PostGIS, pgvector, TimescaleDB 较少
SQL 标准 严格遵循 较多方言
🎯 一句话： PG 像数据库里的瑞士军刀——什么都能干，干得都不差。

1.2 PG 架构概览

┌─────────────────────────────────────────────────────────┐
│                    Client (psql / 应用)                    │
└────────────────────────┬────────────────────────────────┘
                         │
┌────────────────────────▼────────────────────────────────┐
│                   PostgreSQL Server                       │
│  ┌──────────────────────────────────────────────────┐    │
│  │               Postmaster (守护进程)                │    │
│  │  连接管理、fork 子进程处理每个客户端连接            │    │
│  └──────────────────────┬───────────────────────────┘    │
│  ┌──────────────────────▼───────────────────────────┐    │
│  │             共享内存（Shared Buffers）              │    │
│  │  数据缓存、WAL 日志缓冲区、锁信息                  │    │
│  └──────────────────────────────────────────────────┘    │
│  ┌──────────────────────────────────────────────────┐    │
│  │            WAL (Write-Ahead Log)                  │    │
│  │  所有修改先写日志，再写数据文件（崩溃恢复保障）    │    │
│  └──────────────────────────────────────────────────┘    │
└──────────────────────────────────────────────────────────┘

关键设计：

  • 进程模型：每个连接一个进程，隔离性好
  • WAL：所有变更先写日志再写数据文件
  • MVCC：读不阻塞写、写不阻塞读
  • autovacuum：自动清理过期数据版本

2.2 数据库与表

-- 创建数据库
CREATE DATABASE myapp;

-- 创建表
CREATE TABLE users (
    id          BIGSERIAL PRIMARY KEY,
    username    VARCHAR(64) NOT NULL UNIQUE,
    email       VARCHAR(256) NOT NULL,
    password_hash TEXT NOT NULL,
    role        VARCHAR(32) DEFAULT 'user',
    is_active   BOOLEAN DEFAULT TRUE,
    created_at  TIMESTAMPTZ DEFAULT NOW(),
    updated_at  TIMESTAMPTZ DEFAULT NOW(),
    metadata    JSONB DEFAULT '{}',

    CONSTRAINT email_format CHECK (email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$')
);

-- 创建索引
CREATE INDEX idx_users_role ON users(role);
CREATE INDEX idx_users_created ON users(created_at);
CREATE INDEX idx_users_metadata ON users USING GIN(metadata);

2.3 CRUD 操作

-- INSERT
INSERT INTO users (username, email, password_hash, role)
VALUES ('zhangsan', 'zhangsan@example.com', '$2b$12$...', 'admin');

-- INSERT ... RETURNING（PG 特色）
INSERT INTO users (username, email, password_hash)
VALUES ('zhaoliu', 'zhaoliu@example.com', '$2b$12$...')
RETURNING id, created_at;

-- 查询
SELECT * FROM users WHERE role = 'admin' ORDER BY created_at DESC LIMIT 10;

-- 去重计数
SELECT role, COUNT(*) FROM users GROUP BY role;

-- UPDATE ... RETURNING
UPDATE users SET updated_at = NOW() WHERE id = 1 RETURNING *;

-- UPSERT（插入冲突则更新）
INSERT INTO users (username, email, password_hash)
VALUES ('zhangsan', 'newemail@example.com', '$2b$12$...')
ON CONFLICT (username) DO UPDATE
SET email = EXCLUDED.email, updated_at = NOW();

三、高级查询

3.1 JOIN 详解

CREATE TABLE orders (
    id          BIGSERIAL PRIMARY KEY,
    user_id     BIGINT REFERENCES users(id),
    total       NUMERIC(10,2) NOT NULL,
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

INSERT INTO users (id, username) VALUES (1, 'zhangsan'), (2, 'lisi'), (3, 'wangwu');
INSERT INTO orders (id, user_id, total) VALUES (1, 1, 100), (2, 1, 200), (3, 2, 150);

-- INNER JOIN：只有下过订单的用户
SELECT u.username, o.id, o.total
FROM users u INNER JOIN orders o ON u.id = o.user_id;

-- LEFT JOIN：所有用户（没下过单的也会出现）
SELECT u.username, o.id, o.total
FROM users u LEFT JOIN orders o ON u.id = o.user_id;

-- LATERAL JOIN（PG 特色）：对每个用户找最近的一笔订单
SELECT u.username, recent.*
FROM users u
LEFT JOIN LATERAL (
    SELECT id, total FROM orders
    WHERE user_id = u.id
    ORDER BY created_at DESC LIMIT 1
) recent ON TRUE;

3.2 窗口函数

窗口函数让你在不改变行数的情况下做跨行计算。

INSERT INTO orders (user_id, total, created_at) VALUES
    (1, 100, '2024-01-01'), (1, 200, '2024-02-01'),
    (2, 150, '2024-01-15'), (2, 300, '2024-03-01');

-- ROW_NUMBER：给每组内的行编号
SELECT username, total,
    ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY total DESC) as rn
FROM orders o JOIN users u ON u.id = o.user_id;

-- LAG/LEAD：环比分析
SELECT created_at, total,
    LAG(total) OVER (ORDER BY created_at) as prev_total,
    total - LAG(total) OVER (ORDER BY created_at) as diff
FROM orders WHERE user_id = 1;

-- 累计求和
SELECT created_at, total,
    SUM(total) OVER (ORDER BY created_at) as running_total
FROM orders WHERE user_id = 1;
🎯 窗口函数 vs GROUP BY： GROUP BY 折叠行，窗口函数不折叠行。

3.3 CTE 与递归

CTE（公用表表达式）：

WITH active_users AS (
    SELECT id, username FROM users WHERE is_active = TRUE
),
recent_orders AS (
    SELECT user_id, COUNT(*) as order_count
    FROM orders WHERE created_at > NOW() - INTERVAL '30 days'
    GROUP BY user_id
)
SELECT u.username, COALESCE(o.order_count, 0) as recent_orders
FROM active_users u LEFT JOIN recent_orders o ON u.id = o.user_id
ORDER BY recent_orders DESC;

递归 CTE——组织树查询：

CREATE TABLE employees (
    id BIGSERIAL PRIMARY KEY, name VARCHAR(64), manager_id BIGINT REFERENCES employees(id)
);
INSERT INTO employees VALUES
    (1, 'CEO', NULL), (2, 'CTO', 1), (3, 'CFO', 1),
    (5, '后端负责人', 2), (7, '高级工程师', 5), (8, '初级工程师', 7);

-- 找 CTO（id=2）的所有下属
WITH RECURSIVE org_tree AS (
    SELECT id, name, manager_id, 0 as level FROM employees WHERE id = 2
    UNION ALL
    SELECT e.id, e.name, e.manager_id, t.level + 1
    FROM employees e INNER JOIN org_tree t ON e.manager_id = t.id
)
SELECT * FROM org_tree ORDER BY level;

3.4 全文搜索

CREATE TABLE articles (
    id BIGSERIAL PRIMARY KEY, title TEXT, body TEXT,
    tsv TSVECTOR GENERATED ALWAYS AS (
        to_tsvector('simple', coalesce(title, '') || ' ' || coalesce(body, ''))
    ) STORED
);

INSERT INTO articles (title, body) VALUES
    ('PostgreSQL 教程', 'PostgreSQL 是世界上最好的开源数据库'),
    ('RAG 入门指南', 'RAG 是检索增强生成，让 LLM 更准确');

CREATE INDEX idx_articles_tsv ON articles USING GIN(tsv);

SELECT title, ts_rank(tsv, query) as rank
FROM articles, to_tsquery('simple', '数据库') as query
WHERE tsv @@ query ORDER BY rank DESC;
⚠️ 中文分词需安装 zhparser 或 jieba 扩展。

四、索引深入

4.1 B-tree 索引

PG 的默认索引，适用于 =、>、<、BETWEEN、IN、LIKE 'abc%'。

-- 最左前缀原则
CREATE INDEX idx_users_role_created ON users(role, created_at);

WHERE role = 'admin'                            -- ✅ 走索引
WHERE role = 'admin' AND created_at > '2024-01-01'  -- ✅
WHERE created_at > '2024-01-01'                 -- ❌ 跳过最左列

-- 覆盖索引（避免回表）
CREATE INDEX idx_users_role_include ON users(role) INCLUDE (username, email);

4.2 GIN 索引

用于 JSONB、全文搜索、数组等组合类型。

CREATE INDEX idx_users_meta ON users USING GIN(metadata);

-- 走索引的查询：
SELECT * FROM users WHERE metadata @> '{"department": "tech"}';  -- 包含
SELECT * FROM users WHERE metadata ? 'phone';                     -- 存在 key
SELECT * FROM users WHERE metadata @> '{"tags": ["postgresql"]}'; -- 嵌套数组

-- 不走索引的（需要 B-tree）：
SELECT * FROM users WHERE metadata->>'department' = 'tech';

-- 对特定 JSONB 字段建 B-tree：
CREATE INDEX idx_users_lang ON users ((metadata->>'lang'));
WHERE metadata->>'lang' = 'zh'  -- ✅ 走 B-tree

4.3 GiST 与 SP-GiST

-- pg_trgm —— LIKE '%xxx%' 走索引
CREATE EXTENSION pg_trgm;
CREATE INDEX idx_users_username_trgm ON users USING GIN(username gin_trgm_ops);
SELECT * FROM users WHERE username LIKE '%xiao%';  -- ✅ 走索引

-- 范围类型 + 排他约束
CREATE TABLE reservations (
    room_id INT, booked_during DATERANGE,
    EXCLUDE USING GIST (room_id WITH =, booked_during WITH &&)
);
INSERT INTO reservations VALUES (101, daterange('2024-01-01', '2024-01-07'));
INSERT INTO reservations VALUES (101, daterange('2024-01-05', '2024-01-10'));  -- ❌ 冲突

4.4 BRIN 索引

适合天然有序的大表（如日志表），索引只有 B-tree 的 1/1000。

-- B-tree：约 2GB
CREATE INDEX idx_logs_btree ON access_logs(created_at);
-- BRIN：约 2MB
CREATE INDEX idx_logs_brin ON access_logs USING BRIN(created_at) WITH (pages_per_range=32);

4.5 索引选择实战

=, >, <, ORDER BY, LIKE 'abc%'     → B-tree
JSONB 包含/存在查询                  → GIN
全文搜索 tsvector                   → GIN（读多）或 GiST（写多）
数组包含 @>                         → GIN
模糊搜索 LIKE '%xxx%'               → pg_trgm + GIN
超大表 + 有序字段 + 范围查询          → BRIN
几何/空间查询                       → GiST + PostGIS

五、JSONB 专题

5.1 JSONB 基础操作

CREATE TABLE documents (
    id BIGSERIAL PRIMARY KEY, filename VARCHAR(256), meta JSONB DEFAULT '{}'
);

INSERT INTO documents (filename, meta) VALUES (
    '退款政策.pdf',
    '{"pages": 12, "lang": "zh", "tags": ["退款", "政策"], "author": {"name": "张三"}}'::jsonb
);

-- 查询
SELECT meta->>'author' as author_name FROM documents;            -- 文本
SELECT meta->'author'->>'name' as name FROM documents;           -- 链式

-- GIN 索引查询
SELECT * FROM documents WHERE meta @> '{"lang": "zh"}';         -- 包含
SELECT * FROM documents WHERE meta ? 'tags';                     -- 存在 key

-- 更新
UPDATE documents SET meta = jsonb_set(meta, '{version}', '2.2'::jsonb) WHERE id = 1;
UPDATE documents SET meta = meta || '{"reviewer": "李四"}'::jsonb WHERE id = 1;
UPDATE documents SET meta = meta - 'version' WHERE id = 1;       -- 删除键

5.2 JSONB 索引与性能

最佳实践：

  1. 适合存： 标签、配置、元数据（结构不固定的数据）
  2. 不适合存： 核心业务字段（无外键约束、类型约束）
  3. 高频查询字段单独建列： 经常 WHERE 的字段做独立列 + B-tree
  4. 不要把所有东西塞一个 JSONB

6.1 EXPLAIN ANALYZE

-- Seq Scan         全表扫描 ❌

work_mem = 64MB                # 排序/哈希内存

6.3 连接池

PG 是进程模型——1000 个连接 = 1000 个进程。必须用连接池。

# SQLAlchemy 连接池
engine = create_async_engine(
    "postgresql+asyncpg://user:pass@localhost:5432/mydb",
    pool_size=10,             # CPU核数 × 2
    max_overflow=20,
    pool_pre_ping=True,
    pool_recycle=3600,
)

# PgBouncer（服务端连接池）
# 应用→PgBouncer（6432）→PG（5432）
# 2000 应用连接 → 25 PG 连接

七、生产部署

7.1 主从复制

# 主库
psql -c "CREATE ROLE replicator WITH LOGIN REPLICATION PASSWORD 'secret';"
# postgresql.conf: wal_level = replica, max_wal_senders = 5

# 从库
pg_basebackup -h <主库IP> -D /var/lib/postgresql/data -U replicator -P -v -R
systemctl start postgresql

# 验证
psql -c "SELECT * FROM pg_stat_replication;"

7.2 备份策略

# 逻辑备份
pg_dump -h localhost -U postgres -d mydb -Fc > mydb.dump
pg_restore -h localhost -U postgres -d mydb mydb.dump

# 物理备份
pg_basebackup -h localhost -U replicator -D /backup/dir -P -v

# 推荐：每日 pg_dump + 持续 WAL 归档 + 保留 30 天

八、面试题

Q1：PG 和 MySQL 怎么选？

A：PG 适合复杂查询、JSONB、全文搜索；MySQL 适合简单 CRUD。PG 功能更完备、许可证更友好。

Q2：MVCC 是什么？

A：Multi-Version Concurrency Control。每条数据有多个版本，读不阻塞写、写不阻塞读。代价是旧版本需要 VACUUM 清理。

Q3：VACUUM 是什么？

A：回收 MVCC 留下的死元组。autovacuum 自动运行。VACUUM 只回收空间不缩表，VACUUM FULL 缩表但锁表。

Q4：JSONB 和 JSON 的区别？

A：JSON 存文本（每次查询解析→慢），JSONB 存二进制（支持 GIN 索引→快）。永远用 JSONB。

Q5：GIN 和 B-tree 的区别？

A：B-tree 适合等值和范围查询，GIN 适合 JSONB 包含/数组/全文搜索。GIN 更大、更新更慢。

Q6：窗口函数和 GROUP BY 的区别？

A：GROUP BY 折叠行，窗口函数不折叠行——原始行不变，旁边多一列聚合结果。

Q7：递归 CTE 解决什么问题？

A：树形/层级数据遍历（组织架构、分类树）。锚点 UNION ALL 递归部分。

Q8：怎么发现慢查询？

A：pg_stat_statements、慢查询日志（log_min_duration_statement=1000ms）、EXPLAIN ANALYZE。

Q9：为什么 PG 需要连接池？

A：进程模型，每个连接一个进程。连接池把几千应用连接映射到几十个 PG 连接。

Q10：分区表怎么用？

CREATE TABLE orders_partitioned (
    id BIGSERIAL, created_at TIMESTAMPTZ, total NUMERIC(10,2)
) PARTITION BY RANGE (created_at);

CREATE TABLE orders_2024_q1 PARTITION OF orders_partitioned
    FOR VALUES FROM ('2024-01-01') TO ('2024-04-01');
-- 查询只扫对应分区（分区裁剪），删旧数据直接 DROP 子分区（秒级）
